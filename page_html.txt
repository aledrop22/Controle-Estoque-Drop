        function showLoadingOverlay() {
            if (document.getElementById('loading-overlay')) return;

            const filters = document.getElementById('catalog-container');
            const products = document.getElementById('catalog-products');

            const top = filters.offsetTop;
            const height = (products.offsetTop + products.offsetHeight) - filters.offsetTop;

            const rect = filters.getBoundingClientRect();
            const containerLeft = rect.left + window.scrollX;
            const containerWidth = rect.width;

            const overlay = document.createElement('div');
            overlay.id = 'loading-overlay';
            overlay.style.cssText = `
                position: absolute;
                top: ${top}px;
                left: 0;
                width: 100%;
                height: ${height}px;
                background-color: rgba(255, 255, 255, 0.8);
                z-index: 999;
            `;

            // Loader centralizado
            const loader = document.createElement('div');
            loader.style.cssText = `
                position: fixed;
                top: 50%;
                left: ${containerLeft + containerWidth / 2}px;
                transform: translate(-50%, -50%);
                font-size: 18px;
                color: #333;
            `;
            loader.innerHTML = `
                <div class="dt-processing">
                    Carregando<span class="loading-dots">
                        <span></span><span></span><span></span>
                    </span>
                </div>
            `;

            overlay.appendChild(loader);
            document.body.appendChild(overlay);
        }

        function changePerPage() {
            showLoadingOverlay();

            const perPage = document.getElementById('perPage').value;
            const url = new URL(window.location);
            url.searchParams.set('per_page', perPage);
            url.searchParams.set('page', 1); // Resetar para primeira página
            window.location.href = url.toString();
        }

        $(function() {
            $('body').on('click', '.pagination a', function(e) {
                e.preventDefault();
                showLoadingOverlay();

                $('#pagination a').css('color', '#dfecf6');
                // $('#pagination').append('<img style="position: absolute; left: 0; top: 0; z-index: 100000;" src="/images/loading.png" />');

                var url = $(this).attr('href');  
                // getPagination(url);
            
                if($("#name").val() != "" || $("#category_id").val() != "" || $("#iptSorter").val() != ""){                    
                    window.location.href = `${url}&name=${$("#name").val()}&sorter=${$("#iptSorter").val()}&per_page=${$("#perPage").val()}`;
                } else {
                    window.location.href = `${url}&per_page=${$("#perPage").val()}`;
                }                
                // window.history.pushState("", "", url);
            });

            document.getElementById("name").addEventListener("keypress", function(event) {
                if (event.key === "Enter") {
                    event.preventDefault();

                    document.getElementById("search").click();
                }
            });
            
            $('body').on('click', '#search', function(e) {
                e.preventDefault();
                showLoadingOverlay();

                let name = $("#name").val();
                let reserva = $("#reserva").val();
                let category_id = $("#category_id").val();
                let url = 'https://app.seuarmazemdrop.com.br/dashboard/catalog';
                let perPage = $("#perPage").val();

                // getPagination(`${url}?name=${name}&page=1`);
                if($("#name").val() != "" || $("#category_id").val() != ""  || $("#iptSorter").val() != "" || $("#reserva").val() != ""){
                    window.location.href = `${url}?name=${name}&category_id=${category_id}&sorter=${$("#iptSorter").val()}&page=1&reserva=${reserva}&per_page=${perPage}`;
                }else{
                    window.location.href = `${url}?page=1&per_page=${perPage}`;
                }
            });

            $('body').on('click', '.sorter', function(e) {
                e.preventDefault();
                showLoadingOverlay();
                
                let name = $("#name").val();
                let category_id = $("#category_id").val();
                let sorter = e.currentTarget.dataset.sorter;
                let url = 'https://app.seuarmazemdrop.com.br/dashboard/catalog';
                let perPage = $("#perPage").val();

                if($("#iptSorter").val() == sorter){
                    sorter = "";
                    $("#iptSorter").val(sorter)
                }
                
                // getPagination(`${url}?name=${name}&page=1`);
                if($("#name").val() != "" || $("#category_id").val() != "" || sorter != ""){
                    window.location.href = `${url}?name=${name}&category_id=${category_id}&sorter=${sorter}&page=1&per_page=${perPage}`;
                }else{
                    window.location.href = `${url}?page=1&per_page=${perPage}`;
                }
            });

            function getPagination(url) {
                let name = $("#name").val();
                let category_id = $("#category_id").val();

                $.ajax({
                    url : `${url}&name=${name}&category_id=${category_id}`  
                }).done(function (data) {
                    $('#pagination').html(data);  
                }).fail(function () {
                    alert('Skus could not be loaded.');
                });
            }
        });
